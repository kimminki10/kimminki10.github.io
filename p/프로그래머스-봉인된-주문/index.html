def numtoal(num):
    if num <= 25: return chr(97+num)
    n = num % 26
    return numtoal(num // 26 -1) + numtoal(n)

def altonum(al):
    if len(al) <= 1: return ord(al) - ord('a') + 1
    answer = 0
    for i, v in enumerate(al[::-1]):
        answer += (ord(v) - ord('a') +1) * (26 ** i)
    return answer

def solution(n, bans):
    answer = ''
    bs = set([altonum(ban) for ban in bans])
    while True:
        toremove = set()
        for ban in bs:
            if ban <= n:
                n += 1
                toremove.add(ban)
        if not toremove:
            break
        bs -= toremove
    answer = numtoal(n-1)
    return answer
